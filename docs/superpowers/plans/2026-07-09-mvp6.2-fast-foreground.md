# MVP6.2 Fast Foreground Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement MVP6.2 provider-free Fast Interaction output and Fast Foreground Gate so `FAST_ONLY` can produce a replayable, gated foreground text output in the MVP6 debug console.

**Architecture:** Keep the existing MVP5/MVP6 audio-to-routing spine intact and add the ADR-017 foreground layer after committed-turn evidence and before user-visible display. The first slices are provider-free: sync canonical events, add fake Fast Interaction adapter output, implement deterministic gate policy, then integrate with route metadata and the local debug console. Real Fast Interaction provider work is explicitly deferred until the provider-free path proves replay, safety, and no second post-Router model call.

**Tech Stack:** Python 3.11, existing `voice_agent.events`, `voice_agent.adapters`, `voice_agent.runtime`, pytest through `./scripts/test`, Markdown specs under `docs/specs` and `docs/implementation`.

---

```python
from __future__ import annotations

from collections.abc import Mapping, Sequence
from copy import deepcopy
from typing import Any


class FastInteractionValidationError(ValueError):
    """Raised when Fast Interaction output violates ADR-017."""


FOREGROUND_ACTS = frozenset({"ANSWER", "ACK_SLOW", "ACK_PATCH", "SILENCE", "CLARIFY"})
RISK_CLASSES = frozenset({"LOW", "MEDIUM", "HIGH"})
OUTPUT_MODES = frozenset({"real", "mock", "fallback", "degraded"})
BOUNDARY_MARKERS = (
    "booked",
    "paid",
    "deleted",
    "sent the email",
    "confirmed",
    "authorized",
    "tool executed",
)


def validate_fast_interaction_output(output: Mapping[str, Any]) -> dict[str, Any]:
    normalized = deepcopy(dict(output))
    for field in (
        "fast_interaction_output_id",
        "adapter_id",
        "adapter_type",
        "adapter_request_id",
        "turn_id",
        "utterance_id",
        "input_modality",
        "source_event_ids",
        "route_hint",
        "route_prelude",
        "foreground_act",
        "final_fast_evidence_ref",
        "risk_tags",
        "risk_class",
        "confidence",
        "schema_name",
        "normalization_status",
        "output_mode",
        "trace_redaction_level",
    ):
        if field not in normalized or normalized[field] in ("", None):
            raise FastInteractionValidationError(f"missing field: {field}")
    if normalized["adapter_type"] != "fast_interaction":
        raise FastInteractionValidationError("adapter_type must be fast_interaction")
    if normalized["foreground_act"] not in FOREGROUND_ACTS:
        raise FastInteractionValidationError("unsupported foreground_act")
    if normalized["risk_class"] not in RISK_CLASSES:
        raise FastInteractionValidationError("unsupported risk_class")
    if normalized["output_mode"] not in OUTPUT_MODES:
        raise FastInteractionValidationError("unsupported output_mode")
    confidence = normalized["confidence"]
    if not isinstance(confidence, (int, float)) or isinstance(confidence, bool) or not 0 <= confidence <= 1:
        raise FastInteractionValidationError("confidence must be between 0 and 1")
    _validate_candidate_boundary(normalized)
    return normalized


def _validate_candidate_boundary(output: Mapping[str, Any]) -> None:
    candidate = output.get("reply_candidate")
    if not isinstance(candidate, str) or not candidate:
        return
    lowered = candidate.lower()
    risk_tags = tuple(str(tag).lower() for tag in output.get("risk_tags", ()))
    if any(marker in lowered for marker in BOUNDARY_MARKERS) or any(
        tag in {"tool_claim", "side_effect_claim", "confirmation_claim", "current_plan_fact"}
        for tag in risk_tags
    ):
        raise FastInteractionValidationError("candidate boundary violation")
```
